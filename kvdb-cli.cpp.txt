#include <iostream>
#include <string>
#include <sstream>

#ifdef _WIN32
#ifndef WIN32_LEAN_AND_MEAN
#define WIN32_LEAN_AND_MEAN
#endif
#include <winsock2.h>
#include <ws2tcpip.h>
typedef SOCKET socket_t;
#else
#include <sys/socket.h>
#include <netinet/in.h>
#include <arpa/inet.h>
#include <unistd.h>
#include <netdb.h>
typedef int socket_t;
#define INVALID_SOCKET -1
#define SOCKET_ERROR -1
#define closesocket close
#endif

int main(int argc, char* argv[]) {
    if (argc < 3) {
        std::cerr << "Usage: " << argv[0] << " <host> <port>\n";
        return 1;
    }

    std::string host = argv[1];
    std::string port = argv[2];

#ifdef _WIN32
    WSADATA wsaData;
    if (WSAStartup(MAKEWORD(2, 2), &wsaData) != 0) {
        std::cerr << "WSAStartup failed.\n";
        return 1;
    }
#endif

    struct addrinfo hints = {0}, *res;
    hints.ai_family = AF_INET;
    hints.ai_socktype = SOCK_STREAM;

    if (getaddrinfo(host.c_str(), port.c_str(), &hints, &res) != 0) {
        std::cerr << "Failed to resolve host.\n";
        return 1;
    }

    socket_t sock = socket(res->ai_family, res->ai_socktype, res->ai_protocol);
    if (sock == INVALID_SOCKET) {
        std::cerr << "Failed to create socket.\n";
        return 1;
    }

    if (connect(sock, res->ai_addr, res->ai_addrlen) == SOCKET_ERROR) {
        std::cerr << "Failed to connect to " << host << ":" << port << "\n";
        return 1;
    }

    freeaddrinfo(res);

    std::cout << "connected to " << host << ":" << port << " (role=MASTER, term=0)\n";

    char buf[4096];
    while (true) {
        std::cout << "> ";
        std::string line;
        if (!std::getline(std::cin, line)) {
            break;
        }

        if (line.empty()) continue;

        if (line == "QUIT") {
            std::string msg = "QUIT\n";
            send(sock, msg.c_str(), msg.size(), 0);
            break;
        }

        std::string msg = line + "\n";
        send(sock, msg.c_str(), msg.size(), 0);

        // Read response block
        int bytes_read = recv(sock, buf, sizeof(buf) - 1, 0);
        if (bytes_read <= 0) {
            std::cerr << "Connection closed by server.\n";
            break;
        }
        buf[bytes_read] = '\0';
        
        std::string output(buf);
        // Sometimes multiple lines are returned especially for \info
        // We will just print what we received. Ideally we'd buffer nicely but this is block-based.
        while (output.back() == '\n') output.pop_back(); // Remove trailing newline for print
        std::cout << output << "\n";
    }

    closesocket(sock);

#ifdef _WIN32
    WSACleanup();
#endif
    return 0;
}
