#pragma once
#include <string>
#include <unordered_map>
#include <mutex>
#include <memory>
#include <condition_variable>
#include "WALWriter.h"

class KVStore {
public:
    KVStore();
    ~KVStore();

    void init(const std::string& data_dir);

    // Replay/Replication methods
    void set_lsn_and_term(uint64_t lsn, uint64_t term);
    void apply_raw(uint8_t op_type, const std::string& key, const std::string& value);
    bool apply_replicated_record(uint64_t lsn, uint64_t term, uint8_t op_type, const std::string& key, const std::string& value);

    // Client methods
    bool put(const std::string& key, const std::string& value, uint64_t& out_lsn);
    bool del(const std::string& key, uint64_t& out_lsn);
    bool get(const std::string& key, std::string& out_val);
    
    // Server info
    uint64_t get_lsn();
    uint64_t get_term();
    std::string get_data_dir();

    // Metadata management
    void set_node_id(int node_id);
    int get_node_id();
    void set_role(const std::string& role);
    std::string get_role();
    void set_master_addr(const std::string& host, int port);
    std::string get_master_host();
    int get_master_port();
    void set_client_port(int port);
    int get_client_port();

    // Peer progress tracking (Full-duplex ACKs)
    void register_peer(int node_id);
    void remove_peer(int node_id);
    void update_peer_ack(int node_id, uint64_t lsn);
    std::unordered_map<int, uint64_t> get_peer_acks();

    // Notification for replication
    void wait_for_lsn(uint64_t last_seen_lsn);

private:
    std::mutex mu_;
    std::unordered_map<std::string, std::string> map_;
    std::unique_ptr<WALWriter> wal_;
    uint64_t current_lsn_;
    uint64_t current_term_;
    std::string data_dir_;
    std::condition_variable cv_;

    // Node metadata
    int node_id_ = 1;
    std::string role_ = "MASTER"; // MASTER or FOLLOWER
    std::string master_host_ = "";
    int master_port_ = 0;
    int client_port_ = 6001;

    // Replication peer tracking (node_id -> ack_lsn)
    std::unordered_map<int, uint64_t> peer_table_;
};
